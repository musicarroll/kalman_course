\begin{frame}
KF'TAS'DATA'OBS'PROC
\begin{verbatim}
     FUNCTION:

             THIS FUNCTION COMPUTES THE DATA NEEDED TO PROCESS
             CORRECTED AIR DATA VELOCITY OBSERVATIONS FOR
             INERTIAL AIR ALIGNMENT. THE OBSERVATION IS
             FORMED BY INTEGRATING OVER A KALMAN CYCLE THE
             DIFFERENCE BETWEEN AIR DATA VELOCITY(IN THE PLATFORM
             FRAME) AND INERTIAL VELOCITY. THIS FUNCTIONS
             COMPUTES THE MEASUREMENT MATRIX, THE OBSERVATION
             MATRIX AND THE NUMBER OF RESIDUALS.


\end{verbatim}
\end{frame}
