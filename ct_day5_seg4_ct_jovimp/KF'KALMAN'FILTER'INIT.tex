\begin{frame}
KF'KALMAN'FILTER'INIT
\begin{verbatim}
     FUNCTION: The KFKALINT subfunction shall perform filter
     initilization based upon flags from Nav mode control.  The
     initilization shall include the initial covarizance matrix and the
     number of varying states.  KFKALINT shall be capable of handling
     the transition from INS air align to aided INS Navigation.
     KFKALINT shall be capable of performing a variable initialization
     based upon an INS error budget input over the MUX.


\end{verbatim}
