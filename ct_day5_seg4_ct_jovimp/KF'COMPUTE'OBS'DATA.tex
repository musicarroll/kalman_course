\begin{frame}
KF'COMPUTE'OBS'DATA
\begin{verbatim}
     FUNCTION:

             THIS FUNCTION CONTROLS THE COMPUTATION OF THE OBSERVATION
             DATA BY CALLING THE PROCEDURE APPROPRIATE FOR THE
             OBSERVATION TO BE PROCESSED THIS KALMAN CYCLE. THE OUTPUTS
             ARE THE NUMBER OD RESIDUALS, THE OBSERVATION RESIDUALS
             THE OBSERVATION NOISE VECTOR, AND THE VARYING AND
             NON-TIME VARYING OBSERVATION MATRICES.


\end{verbatim}
\end{frame}
