\begin{frame}
KF'TILT'BIAS'CORRELATION
\begin{verbatim}
     FUNCTION:

              THIS FUNCTION DETERMINES THE CORRELATION
              OF THE TILT AND ACCELEROMETER BIAS FOLLOWING
              GROUND ALIGNMENT OF AN INU. THE PROCESSING
              SHALL BE INDEPENDENT OF GIMBAL/STRAPDOWN
              IMPLEMENTATION USING THE SENSOR FRAME MATRIX
              FROM THE SNU I10 MESSAGE.

\end{verbatim}
\end{frame}
