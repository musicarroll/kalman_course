\begin{frame}
KF'MOTION'ALIGN'STATUS
\begin{verbatim}
     FUNCTION:

          THIS ROUTINE MONITORS THE HEADING COVARIANCE TO DETERMINE
          THE STATUS OF MOTION ALIGNMENT, AS WELL AS THE POINT AT
          WHICH THE NAVIGATION MODE CAN BE ENTERED (COARSE READY)
          AND THE FULL ACCURACY STATE (NAV READY). AT THE END OF TILT
          COARSE ALIGN, THE COARSE ALIGN COMPLETE INDICATION IS SET.
          A NAV READY INDICATION IS SET AT THE NAV READY CONDITION.
          THE COARSE ALIGN COMPLETE IS SET AFTER BOTH TILT COVARIANCES
          ARE BELOW 50 ARC SECONDS, AND THE RESIDUAL TILTS ARE BELOW
          50 ARC SECONDS.

          DEGRADED MOTION ALIGN COMPLETE IS BASED ON THE X AND Y
          TILTS LESS THAN 25 ARC SECONDS AND THE HEADING COVARIANCE
          GREATER THAN 1.0 MRAD SQUARED. FULL ACCURARCY TRANSFER
          ALIGN COMPLETE IS BASED ON HEADING COVARIANCES LESS THAN
          1.O MRAD SQUARED.



\end{verbatim}
\end{frame}
