\begin{frame}
KF'GEO'POSITION'FIX'OBS'PROC
\begin{verbatim}
     FUNCTION:
             THIS FUNCTION COMPUTES ALL THE REQUIRED INFORMATION
             FOR BOTH VISUAL AND MMR POSITION FIXES USING DIFFERENT
             NOISE PARAMETERS. FUTHER THE NOISE IS MADE RANGE
             DEPENDENT.THE FOLLOWING IS Y'KAL'OBS DATA FORMAT
 
                  Y'KAL'OBS(1)   = NORTH POSITION
                  Y'KAL'OBS(2)   = EAST POSITION
                  Y'KAL'OBS(3)   = COS ALPHA
                  Y'KAL'OBS(4)   = SIN ALPHA
                  Y'KAL'OBS(5)   = H
                  Y'KAL'OBS(6)   = RANGE
                  Y'KAL'OBS(7)   = OBSERVATION TYPE
                  Y'KAL'OBS(8)   = BAD/GOOD POS FIX
 
\end{verbatim}
\end{frame}
