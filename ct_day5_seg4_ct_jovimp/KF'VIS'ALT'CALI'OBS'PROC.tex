\begin{frame}
KF'VIS'ALT'CALI'OBS'PROC
\begin{verbatim}
     FUNCTION:
             THIS FUNCTION COMPUTES ALL THE REQUIRED INFORMATION
             PERTINENT TO THE VIS/ALT CALIBRATION. Y'KAL'OBS
             CONTAINS THE FOLLOWING DATA

                  Y'KAL'OBS(1)   = EAST ERROR
                  Y'KAL'OBS(2)   = NORTH ERROR
                  Y'KAL'OBS(3)   = COS ALPHA
                  Y'KAL'OBS(4)   = SIN ALPHA
                  Y'KAL'OBS(5)   = H
                  Y'KAL'OBS(6)   = RANGE
                  Y'KAL'OBS(7)   = OBSERVATION TYPE
                  Y'KAL'OBS(8)   = BAD/GOOD POS FIX
                  Y'KAL'OBS(9)   = DELTA H
                  Y'KAL'OBS(10)  = BAD/GOOD ALTITUDE FIX

\end{verbatim}
\end{frame}
