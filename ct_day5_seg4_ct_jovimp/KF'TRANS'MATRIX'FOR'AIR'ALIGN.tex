\begin{frame}
KF'TRANS'MATRIX'FOR'AIR'ALIGN
\begin{verbatim}
     FUNCTION:
             THIS FUNCTION CALCUALTES THE NON-ZERO FVV AND FVN MATRIX
             ELEMENTS FOR THE COARSE AIR ALIGN KALMAN FILTER. THE
             FORMS A TRANSITION MATRIX FOR CALCULATIONS IN THE
             PLATFORM REFERENCE FRAME OF THE ALIGNING INS RATHER
             THAN THE COMPUTER FRAME USED FOR AIDED NAVIOGATION.
             THE EQUATION MECHANIZED IN THIS ROUTINE ARE
             ALLPICABLE TO BOTH GIMBALED AND STRAPDOWN ALLPICATIONS.
             THE B MATRIX SENSOR FRAME IDENTITY FOR GIMBALLED
             SYSTEMS AND GIVE THE SENSOR ORIENTATION FOR STRAPDOWN
             SYSTEMS, BUT THIS IS HANDLED IN ACCUM.

\end{verbatim}
\end{frame}
