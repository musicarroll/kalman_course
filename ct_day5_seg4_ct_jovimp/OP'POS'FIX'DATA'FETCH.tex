\begin{frame}
OP'POS'FIX'DATA'FETCH
\begin{verbatim}
     FUNCTION:  This unit checks the observation validity flag for
     each type of position fix to determine the availability of the 
     position fix data.  If the data is available, this unit transfers
     the data into the observation data table, attaches a system time
     tag, and sets the corresponding observation flag to true.




\end{verbatim}
\end{frame}
