\begin{frame}
KF'INU'COVARIANCE'CAL
\begin{verbatim}
     FUNCTION:

          THIS ROUTINE CALCULATES THE REQUIRED COVARIANCES FOR THE
          INU FILTERS FOR USE BY BASIC NAVIGATION AND THE CEP
          CALCULATION. THESE INCLUDE:

                 INU LATITUDE COVARIANCE
                 INU LONGITUDE COVARIANCE
                 INU LATITUDE LONGITUDE CROSS COVARIANCE
                 INU HEADING COVARIANCE
                 INU X-TILT  COVARIANCE
                 INU Y-TILT  COVARIANCE
                 INU ALTITUDE COVARIANCE
                 INU X'VELOCITY COVARIANCE
                 INU Y'VELOCITY COVARIANCE
                 INU Z'VELOCITY COVARIANCE
                 INU THETA X THETA Y CROSS COVARIANCE
                 INU DVS X SCALE FACTOR COVARIANCE
                 INU DVS Y BORESIGHT COVARIANCE
                 INU DVS Z BORESIGHT COVARIANCE
                 INU DVS X BIAS COVARIANCE
                 INU DVS Y BIAS COVARIANCE
                 INU DVS Z BIAS COVARIANCE


\end{verbatim}
\end{frame}
