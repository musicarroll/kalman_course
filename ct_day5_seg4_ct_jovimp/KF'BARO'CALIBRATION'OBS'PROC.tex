\begin{frame}
KF'BARO'CALIBRATION'OBS'PROC
\begin{verbatim}
     FUNCTION:
             THIS FUNCTION COMPUTES ALL THE REQUIRED INFORMATION
             PERTINENT TO THE RADAR ALTIMETER BARO-ALTIMETER CALIBRATION.
             THE OBSERVATION RESIDUAL IS OF THE FORM
                       Y(1) = HC - (HR + HTER)
             WHERE HC IS THE CORRECTED INU BARO INERTIAL ALTITUDE,
             HR IN THE RADAR OUTPUT AND HTER IS THE REFERENCE ALTITUDE
             ABOVE THE ELLIPSOID.


\end{verbatim}
\end{frame}
