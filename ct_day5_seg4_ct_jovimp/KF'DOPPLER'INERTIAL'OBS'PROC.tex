\begin{frame}
KF'DOPPLER'INERTIAL'OBS'PROC
\begin{verbatim}
     FUNCTION:
          THE DVSDAT PERFORMS DOPPLER OBSERVATION PROCESSING.  THE
          INPUTS ARE THE CORRECTED INU VELOCITIES AT THE END OF THE
          DOPPLER DATA ACCUMULATION PERIOD, AND INTERGRALS FOR THE
           VEHICLE B MATRIX, AND THE B MATRIX TIME BODY VELOCITY (AS
           FOUND IN OPHACDAT).  THE USE OF CORRECTED INU VELOCITY IN
           FORMING Y(1), Y(2) & Y(3) WILL ACCOUHNT FOR RESIDUAL ERROR
           EXTRAPOLATION NOT APLLIED TO THE INU.

           FOR COARSE ALIGN, Y'KAL'OBS (4) = 1, THE H MATRIX WILL
           CORRESPOND TO PLATFORM REFERENCE COORDINATES. FOR FINE ALIGN,
           Y'KAL'OBS (4) = 0, THE H MATRIX WILL CORRESPOND TO COMPUTER
           REFERENCE COORDINATES.

\end{verbatim}
\end{frame}
