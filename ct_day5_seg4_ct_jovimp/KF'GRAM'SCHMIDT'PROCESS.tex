\begin{frame}
KF'GRAM'SCHMIDT'PROCESS
\begin{verbatim}
     FUNCTION:

       THIS FUNCTION PRODUCES THE TIME EXTRAPOLATED U-D FACTORS
       USING THE MODIFIED GRAM-SCHMIDT ORTHOGONALIZATION ALGOTHRIM.
       THE INPUTS ARE THE CONCATENATED MATRIX CAT'F'U =(FU:I) AND THE
       CONCATENATED GIAGONAL MATRIX DIAG(D,Z) STORED IN VECTOR
       FORM IN THE ARRAY CAT'D'Z. THE OUTPUTS IS THE VECTOR STORED
       EXTRAPOLATED U-D FACTORS.


\end{verbatim}
\end{frame}
