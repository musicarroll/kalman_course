\begin{frame}
KF'SPARSE'MAT'VECT'MULT
\begin{verbatim}
     FUNCTION:
          THIS ROUTINE PERFORMS MATRIX-VWCTOR MULTIPLICATION
          UTILIZING THE SPARSE MATURE OF THE INPUT MATRIX.
          THE SPARES MATRIX IS STORED IN THE INPUT ARRAY
          "SV'FEL" AND THE INDICES OF THE NON-ZERO VALUES
          IN THE INPUT ARRAYS "SV'IROW" AND "SV'JCOL".



\end{verbatim}
