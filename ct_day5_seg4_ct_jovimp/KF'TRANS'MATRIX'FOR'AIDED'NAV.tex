\begin{frame}
KF'TRANS'MATRIX'FOR'AIDED'NAV
\begin{verbatim}
     FUNCTION:

             THIS FUNCTION CALCUALTES THE NON-ZERO FVV AND FVN MATRIX
             ELEMENTS FOR THE AIDED INERTIAL KALMAN FILTER AND FINE
             MOTION ALIGNMENT. THE STATE VECTOR IS DEFINED IN THE
             COMPUTER FRAME OF THE INS. THE TRANSITION MATRIX IS
             USEABLE FOR BOTH GIMBAL AND STRAPDOWN APPLICATIONS.
             THE B MATRIX DETERMINES IMPLICITLY THE CASE. IF A
             GIMBALLED SYSTEM IS USED, B IS THE IDENTY MATRIX. IF
             STRAPDOWN SYSTEM IS USED B DETERMINES THE SENSOR
             OIRIENTATION RELATIVE TO THE NAVIGATION REFERENCE
             FRAME.

\end{verbatim}
\end{frame}
