\begin{frame}
KF'EXT'ERROR'STATE'AIR'ALIGN
\begin{verbatim}
     FUNCTION:
             THIS FUNCTION PERFORMS AIDED INS AIR ALIGNMENT
             EXTRAPOLATION TO THE PRESENT TIME IN ORDER TO
             APPLY THE ERROR STATES TO THE INS. THIS ROUTINE
             CALCULATES ERROR DATA FOR THE INS ERROR EXTRAPOLATION
             EQUATIONS FROM WHICH F02 CORRECTIONS VECTOR
             IS GERERATED. THE HEADING ERROR IS NOT APPLIED
             DURING ALIGNMENT. IT IS APPLIED AT THE END OF
             ALIGNMENT.

\end{verbatim}
\end{frame}
