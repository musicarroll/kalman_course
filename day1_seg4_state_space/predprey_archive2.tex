			}
			\onslide<6->{

			}
			\onslide<7-> {
				\item Genetic adaptation is negligible
			}

\begin{frame}

	\onslide<1->{
		\frametitle{Coupled ODEs}
		\framesubtitle{Predator-Prey State Space Model (Equations)}
	}
	\onslide<2->{
		State space model for Lotka-Volterra predator-prey system:
		
		\[
		\frac{dR}{dt} = r_R R - c_{RW} R W
		\]
		
		\[
		\frac{dW}{dt} = c_{WR} R W - d_W W
		\]
}
	\onslide<3->{
		\small
		Where:
		\begin{align*}
		R & : \text{Prey (Rabbits) population} \\
		W & : \text{ Predator (Wolves) population} \\
		r_R & : \text{Prey intrinsic growth rate} \\
		c_{RW} & : \text{Predation rate, represents the rate}\\ 
		& \;\;\;\; \text{at which predators capture and consume prey} \\
		c_{WR} & : \text{Conversion efficiency, represents the efficiency}\\ 
		 & \;\;\;\; \text{ with which consumed prey are converted into new predators} \\
		d_W & : \text{Predator death rate}
		\end{align*}
    }
	\onslide<4->{\center --------------------------------------------------------------------}

\end{frame}


	
	\begin{frame}
		\onslide<1->{
		\frametitle{Coupled ODEs}
		\framesubtitle{Predator-Prey State Space Model (Matrix Form)}
	}
	\onslide<2->{
		In matrix form, the state space model can be expressed as:
		\[
		\left[\begin{array}{c}
		\frac{dR}{dt} \\
		\frac{dW}{dt}
		\end{array}\right]
		=
		\left[\begin{array}{cc}
		r_R & -c_{RW} \\
		c_{WR} & -d_W
		\end{array}\right]
		\left[\begin{array}{c}
		R \\
		W
		\end{array}\right]
		\]
}

}

	\end{frame}
