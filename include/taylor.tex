\newcommand{\mvartaylortermsym}[3]
{
D^{#1}_{#2,#2_0}#3
}
%args: {k}{x}{f}{x_0}
\newcommand{\mvartaylorterm}[4]
{
\left(\displaystyle \sum_{i=1}^n(#2_i-[#4]_i)\pderiv{#3}{#2_i}\Bigr|_{\substack{#4}}\right)^#1
}
\newcommand{\mvartaylortermsymdef}[4]
{
\mvartaylortermsym{#1}{#2}{#3}:=\mvartaylorterm{#1}{#2}{#3}{#4}
}
\newcommand{\mvartaylorseries}[2]
{
#2(#1)=#2(#1_0)+\mvartaylortermsym{1}{#1}{#2}+\dfrac{1}{2!}\mvartaylortermsym{2}{#1}{#2}
	+\dfrac{1}{3!}\mvartaylortermsym{3}{#1}{#2}+\cdots
}
% args:  f x_0  X
\newcommand{\Dopsumtaylor}[3]
{
#1\left(#2_0\right)+\Dopnth{1}{#3,#2}{#1}
  +\frac{1}{2 !}\Dopnth{2}{#3,#2}{#1}   + \cdots
}
%args: {k}{x}{f}{x_0}
\newcommand{\svartaylorterm}[4]
{
 (#2-#4)^{#1}\derivnth{#3}{#2}{#1}\Bigr|_{\substack{#4}}
}
\newcommand{\Dopsumtayloralt}[4]
{
#1\left(#2_0\right)+\frac{1}{2!}\Dop{#1}{#2}\left(#2_0\right)#4
   +\cdots
   +\frac{1}{n!}\Dopnth{#1}{#2}{#3}\left(#2_0\right){#3 !}#4^{#3}
}
\newcommand{\tayloropdef}[4]
{
\mvartaylortermsym{#1}{#2}{#3}:=
\begin{cases}
	\svartaylorterm{#1}{#2}{#3}{#2_0}, &\text{if dim$(x)=1$;}\\
	\mvartaylorterm{#1}{#2}{#3}{#2_0}, &\text{if dim$(x)>1$.}
\end{cases}
}
