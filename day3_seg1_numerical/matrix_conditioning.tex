\begin{frame}
Matrix Conditioning
\begin{itemize}
  \item Ill-Conditioned vs. Well-Conditioned
  \item Condition number of matrix:
\[\text{cond}(M)=\dfrac{\lambda_{\text{max}}}{\lambda_{\text{min}}}\]
where $\lambda_{\text{max}}$ is maximum eigenvalue of $M$, while $\lambda_{\text{max}}$ is the minimum eigenvalue of $M$.
\end{itemize} 
\end{frame}

\begin{frame}
Matrix Conditioning
\begin{itemize}
  \item If $\text{cond}(M)\approx 1$, then $M$ is well-conditioned for matrix inversion
  \item If $\text{cond}(M)>>1$, then $M$ is ill-conditioned for inversion
\end{itemize} 
\end{frame}
