\begin{frame}
CT I Kalman Filter: Rate Group Drilldown
\begin{itemize}
   \item Primary Kalman Cycle: Every 10 seconds or .1 Hz
	\begin{itemize}
		\item Final extrapolation and application of observation update
	 	\item Doppler compensation (weighted averages)
		\item Kalman initialization
		\item Buffering of Kalman data from 4 Hz and 1 Hz
		\item Mag Var initialization
		\item Weighted averaging
	\end{itemize}
\end{itemize} 
\end{frame}
